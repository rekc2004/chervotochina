table inet nat8 {
	set seen {
		type ipv4_addr
		size 65535
		flags dynamic,timeout
		timeout 2m
	}

	chain pre {
		type nat hook prerouting priority dstnat; policy accept;
		iifname "wan" udp dport 4433 ip saddr @seen dnat ip to 192.168.10.2:4433
	}

	chain post {
		type nat hook postrouting priority srcnat; policy accept;
		oifname "wan" ip saddr 192.168.10.0/24 masquerade
	}

	chain fwd8 {
		type filter hook forward priority filter; policy drop;
		iifname "lan" oifname "wan" udp sport 4433 update @seen { ip daddr }
		iifname "lan" oifname "wan" accept
		iifname "wan" ct state established,related accept
		iifname "wan" ct status dnat accept
	}

	chain inp8 {
		type filter hook input priority filter; policy accept;
		iifname "wan" drop
	}
}
